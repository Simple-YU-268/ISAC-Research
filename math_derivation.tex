\section{原始问题}
%==============================================================================

\begin{align}
\min_{\substack{\{\vect{w}_{m,k}\}, \{\mat{Z}_m\}, \\ \{b_{mp}\}}} \quad 
& \sum_{m=1}^{M} \left( \sum_{k=1}^{K} \norm{\vect{w}_{m,k}}^2 + \tr(\mat{Z}_m) \right) \tag{P1} \\
\text{s.t.} \quad 
& \text{SINR}_k^{\text{wc}} \geq \gamma_k, \quad \forall k \in \mathcal{K} \tag{P1-C1} \\
& \text{SINR}_{S,p} := \frac{|\vect{g}_p^H \mat{Z} \vect{g}_p|}{\sigma_s^2} \geq \gamma_S^{\text{PoD}}, \quad \forall p \in \mathcal{P} \tag{P1-C2} \\
& \tr(\mat{J}_p^{\text{data}}) \geq \Gamma_{\text{Track}, p}, \quad \forall p \in \mathcal{P} \tag{P1-C3} \\
& \sum_{m=1}^{M} b_{mp} = N_{\text{req}}, \quad \forall p \in \mathcal{P}^{\text{active}} \tag{P1-C4} \\
& \sum_{k=1}^{K} \norm{\vect{w}_{m,k}}^2 + \tr(\mat{Z}_m) \leq P_{\max}, \quad \forall m \in \mathcal{M} \tag{P1-C5} \\
& \mat{Z}_m \succeq \mat{0}, \quad \forall m \in \mathcal{M} \tag{P1-C6} \\
& b_{mp} \in \{0, 1\}, \quad \forall m \in \mathcal{M}, p \in \mathcal{P} \tag{P1-C7} \\
& \rank(\mat{W}_k) = 1, \quad \forall k \in \mathcal{K} \tag{P1-C8}
\end{align}

其中 $\mat{W}_k = \vect{w}_k \vect{w}_k^H$ 为协作通信协方差矩阵，$\vect{w}_k = [\vect{w}_{1,k}^T, \ldots, \vect{w}_{M,k}^T]^T \in \mathbb{C}^{MN_t}$ 为所有 AP 到用户 $k$ 的波束协作合并向量（每 AP 独立约束在 per-AP 功率 (P1-C5) 中，详见 (P1-C5) 注）。最坏情况 SINR 定义为：
\begin{equation}
\text{SINR}_k^{\text{wc}} = \min_{\norm{\Delta\vect{h}_k} \leq \epsilon_h \norm{\hat{\vect{h}}_k}} \frac{|(\hat{\vect{h}}_k + \Delta\vect{h}_k)^H \vect{w}_k|^2}{\sum_{j\neq k}|(\hat{\vect{h}}_k + \Delta\vect{h}_k)^H \vect{w}_j|^2 + \sigma_c^2}
\end{equation}

%==============================================================================
\section{非凸性分析}
%==============================================================================

\subsection{目标函数}

目标函数 $\sum_{m,k} \|\vect{w}_{m,k}\|^2 + \sum_m \tr(\mat{Z}_m)$ 是平方范数与迹之和。每个项 $\|\vect{w}_{m,k}\|^2 = \vect{w}_{m,k}^H \vect{w}_{m,k}$ 是凸二次函数（正定 Hessian）。但与秩一约束 (P1-C8) 结合后，可行集变为非凸。

\textbf{凸性状态}：目标本身凸，但整体问题因约束而非凸。

\subsection{通信 SINR 约束 (P1-C1)}

\begin{equation}
\frac{|\vect{h}_k^H \vect{w}_k|^2}{\sum_{j\neq k}|\vect{h}_k^H \vect{w}_j|^2 + \sigma_c^2} \geq \gamma_k
\end{equation}

这是广义分式规划约束。函数 $f(\vect{w}) = \frac{|\vect{h}^H \vect{w}|^2}{\sum |\vect{h}^H \vect{w}_j|^2 + \sigma^2}$ 单独对每个 $\vect{w}_k$ 拟凸（固定其他变量），但对 $\{\vect{w}_k\}_{k=1}^K$ 联合非凸（分子分母变量间双线性耦合）。

\textbf{非凸原因}：超水平集 $\{ \vect{w} : f(\vect{w}) \geq \gamma \}$ 非凸。对满足约束的 $\vect{w}^{(1)}, \vect{w}^{(2)}$，其凸组合 $\theta \vect{w}^{(1)} + (1-\theta) \vect{w}^{(2)}$ 通常违反 SINR 约束（分母交叉项）。

\textbf{非凸源}：NC1（SINR 分式结构）+ NC2（最坏情况 CSI 不确定性导致半无限约束）。

\subsection{感知 SINR 约束 (P1-C2)}

\begin{equation}
\frac{|\vect{g}_p^H \mat{Z} \vect{g}_p|}{\sigma_s^2} \geq \gamma_S^{\text{PoD}}
\end{equation}

对 $\mat{Z}$ 看似线性，但 $\mat{Z}$ 通过功率约束 (P1-C5) 与 $\{\vect{w}_k\}$ 耦合，使联合可行集非凸。

\subsection{PCRB 约束 (P1-C3)}

FIM 迹涉及依赖于 $\mat{R}_X$ 的矩阵逆：
\begin{equation}
\tr\left( \left[ \mat{J}_p^{\text{prior}} + \mat{J}_p^{\text{data}}(\mat{R}_X) \right]^{-1} \right) \leq \Gamma_{\text{Track},p}
\end{equation}

\textbf{非凸原因}：矩阵逆 $\mat{X}^{-1}$ 在 $\mat{X} \succ \mat{0}$ 上凸（矩阵凸），但迹的逆是凸函数。此处是凸函数的下界约束，对应凸函数的上界——这是非凸约束。

等价地，$\tr(\mat{J}_p^{\text{data}}) \geq \Gamma_{\text{Track},p}$ 对 $\mat{J}_p^{\text{data}}$ 线性，但 $\mat{J}_p^{\text{data}}$ 是 $\mat{R}_X$ 的非线性函数（信道梯度二次型）。

\textbf{非凸源}：NC6（矩阵逆 / FIM 非线性依赖）。

\subsection{AP 选择约束 (P1-C4)}

\begin{equation}
\sum_{m=1}^{M} b_{mp} = N_{\text{req}}, \quad b_{mp} \in \{0, 1\}
\end{equation}

\textbf{非凸原因}：集合 $\{0, 1\}$ 是离散两点集。凸包为 $[0, 1]$，但约束要求整数值。

\textbf{非凸源}：NC3（二进制组合）。

\subsection{功率约束 (P1-C5)}

\begin{equation}
\sum_{k=1}^{K} \|\vect{w}_{m,k}\|^2 + \tr(\mat{Z}_m) \leq P_{\max}
\end{equation}

\textbf{凸性状态}：凸约束（对 $\vect{w}_{m,k}$ 二阶锥约束，对 $\mat{Z}_m$ 线性）。但耦合通信与感知变量，与其他约束结合后贡献整体非凸性。

\textbf{状态}：已凸，无需变换。

\subsection{PSD 约束 (P1-C6)}

$\mat{Z}_m \succeq \mat{0}$ 定义凸锥（半正定锥）。

\textbf{状态}：已凸，无需变换。

\subsection{二进制约束 (P1-C7)}

$b_{mp} \in \{0, 1\}$ 与 (P1-C4) 同非凸性。

\subsection{秩一约束 (P1-C8)}

$\rank(\mat{W}_k) = 1$，其中 $\mat{W}_k = \vect{w}_k \vect{w}_k^H$。

\textbf{非凸原因}：秩一 PSD 矩阵集合为：
\begin{equation}
\mathcal{F}_{\text{rank-1}} = \{ \mat{W} \succeq \mat{0} : \rank(\mat{W}) = 1 \}
\end{equation}

这是非凸流形。例如 $\mat{W}^{(1)} = \vect{v}_1 \vect{v}_1^H$ 和 $\mat{W}^{(2)} = \vect{v}_2 \vect{v}_2^H$（$\vect{v}_1 \perp \vect{v}_2$），其平均 $\frac{1}{2}(\mat{W}^{(1)} + \mat{W}^{(2)})$ 秩为 2，违反约束。

\textbf{非凸源}：NC4（秩一要求）。

\section{逐步凸化}
%==============================================================================

\subsection{Step 1: 协方差矩阵提升（消除 NC1, NC5）}

\textbf{变换}：将每 AP 的向量变量提升到全局矩阵空间。

原始变量：$\vect{w}_{m,k} \in \mathbb{C}^{N_t}$，$\mat{Z}_m \in \mathbb{H}^{N_t}_+$。

定义 AP 协作合并波束 $\vect{w}_k \in \mathbb{C}^{MN_t}$ 与每 AP 选择矩阵 $\mat{E}_m \in \mathbb{R}^{MN_t \times MN_t}$：

\begin{align}
\vect{w}_k &\triangleq [\vect{w}_{1,k}^T, \vect{w}_{2,k}^T, \ldots, \vect{w}_{M,k}^T]^T \in \mathbb{C}^{MN_t} \\
\mat{E}_m &\triangleq \text{diag}(\underbrace{0,\ldots,0}_{(m-1)N_t}, \underbrace{1,\ldots,1}_{N_t}, \underbrace{0,\ldots,0}_{(M-m)N_t}) \in \mathbb{R}^{MN_t \times MN_t}
\end{align}

则 $\vect{E}_m \vect{w}_k = [\vect{0}^T, \vect{w}_{m,k}^T, \vect{0}^T]^T$（仅取 AP $m$ 的分量），每 AP 功率 $\sum_k \|\vect{w}_{m,k}\|^2 + \tr(\mat{Z}_m) = \tr(\mat{E}_m \mat{R}_X)$，其中

\begin{equation}
\mat{R}_X \triangleq \sum_{k=1}^K \mat{W}_k + \mat{Z}, \quad \mat{W}_k = \vect{w}_k \vect{w}_k^H \in \mathbb{H}^{MN_t}, \quad \mat{Z} = \sum_{m=1}^M \mat{Z}_m \in \mathbb{H}^{MN_t} \tag{S1.1}
\end{equation}

\textbf{作用}：
\begin{itemize}
    \item 目标：$\|\vect{w}_k\|^2 = \tr(\mat{W}_k)$ 对 $\mat{W}_k$ 线性化。
    \item 功率：$\sum_k \|\vect{w}_{m,k}\|^2 + \tr(\mat{Z}_m) \leq P_{\max}$ 变为 $\tr(\mat{E}_m \mat{R}_X) \leq P_{\max}$（对 $\mat{W}_k, \mat{Z}$ 线性）。
    \item SINR：$|\vect{h}_k^H \vect{w}_k|^2 = \tr(\vect{h}_k \vect{h}_k^H \mat{W}_k)$ 对 $\mat{W}_k$ 线性化。
\end{itemize}

\textbf{等价性}：严格等价（$\vect{w}_k$ 与秩一 $\mat{W}_k$ 一一映射；每 AP 功率与 $\tr(\mat{E}_m \mat{R}_X)$ 也一一对应）。

\textbf{剩余非凸性}：秩一约束 $\rank(\mat{W}_k) = 1$ 仍存在（NC4）。

\subsection{Step 2: 半定松弛（消除 NC4）}

\textbf{变换}：将秩一约束松弛为 PSD 锥。

原始：$\mat{W}_k \succeq \mat{0}$，$\rank(\mat{W}_k) = 1$。

松弛后：仅 $\mat{W}_k \succeq \mat{0}$。

\textbf{可行集变化}：
\begin{equation}
\mathcal{F}_{\text{rank-1}} = \{\mat{W} \succeq \mat{0} : \rank(\mat{W}) = 1\} \subset \mathcal{F}_{\text{SDR}} = \{\mat{W} \succeq \mat{0}\}
\end{equation}

\textbf{等价性类型}：紧致松弛。

\begin{proposition}[SDR 紧致性条件]
\begin{enumerate}
    \item 若 $K \leq 2$，SDR 紧致：最优 $\mat{W}_k^*$ 满足 $\rank(\mat{W}_k^*) = 1$。
    \item 若 $K > 2$，SDR 提供下界：$P_{\text{SDR}}^* \leq P_{\text{original}}^*$。
    \item 高斯随机化以 $O(1/L)$ 性能损失恢复可行秩一解。
\end{enumerate}
\end{proposition}

\textbf{影响}：目标值可能降低（松弛扩大可行集）。$K > 2$ 时获得原问题下界。

\subsection{Step 3: S-Procedure 处理鲁棒 SINR（消除 NC2）}

\textbf{变换}：将半无限最坏情况约束转化为有限 LMI。

原始（半无限）：
\begin{equation}
\min_{\norm{\Delta\vect{h}_k} \leq \epsilon_h} \frac{|(\hat{\vect{h}}_k + \Delta\vect{h}_k)^H \mat{W}_k (\hat{\vect{h}}_k + \Delta\vect{h}_k)|}{\sum_{j\neq k}|(\hat{\vect{h}}_k + \Delta\vect{h}_k)^H \mat{W}_j(\hat{\vect{h}}_k + \Delta\vect{h}_k)| + \sigma_c^2} \geq \gamma_k
\end{equation}

定义 $\mat{A}_k = \frac{1}{\gamma_k} \mat{W}_k - \sum_{j\neq k} \mat{W}_j$。由 S-Procedure（球上二次约束充要条件，含 $\mu_k$ 松弛变量），半无限约束等价于：

\begin{equation}
\exists \mu_k \geq 0 : \begin{bmatrix} \mat{A}_k + \mu_k \mat{I} & \mat{A}_k \hat{\vect{h}}_k \\[1.2ex] \hat{\vect{h}}_k^H \mat{A}_k & \hat{\vect{h}}_k^H \mat{A}_k \hat{\vect{h}}_k - \sigma_c^2 - \mu_k \epsilon_h^2 \norm{\hat{\vect{h}}_k}^2 \end{bmatrix} \succeq \mat{0} \tag{S3.1}
\end{equation}

\textbf{详细推导}：将 (P1-C1) 改写为关于 $\Delta\vect{h}_k$ 的二次不等式 $\tilde{f}_1(\Delta\vect{h}_k) \geq 0$，其中

\begin{align}
\tilde{f}_1(\Delta\vect{h}) &\triangleq (\hat{\vect{h}} + \Delta\vect{h})^H \mat{A}_k (\hat{\vect{h}} + \Delta\vect{h}) - \sigma_c^2 \notag \\
&= \Delta\vect{h}^H \mat{A}_k \Delta\vect{h} + 2\Real\{\hat{\vect{h}}^H \mat{A}_k \Delta\vect{h}\} + \hat{\vect{h}}^H \mat{A}_k \hat{\vect{h}} - \sigma_c^2
\end{align}

不确定集由 $\tilde{f}_2(\Delta\vect{h}) \triangleq \epsilon_h^2 \norm{\hat{\vect{h}}}^2 - \norm{\Delta\vect{h}}^2 \geq 0$ 描述（相对误差界 $\norm{\Delta\vect{h}} \leq \epsilon_h \norm{\hat{\vect{h}}}$）。由 S-Procedure

\begin{quote}
``存在 $\mu_k \geq 0$ 使 $\tilde{f}_1(\Delta\vect{h}) - \mu_k \tilde{f}_2(\Delta\vect{h}) \geq 0, \forall \Delta\vect{h}$'' \quad $\Leftrightarrow$ \quad 上述 LMI (S3.1)
\end{quote}

（充分性显然；必要性由 $\mat{A}_k + \mu_k \mat{I} \succeq \mat{0}$ 与 Lagrangian 对偶推导保证，参见 Luo 2010 推论 3.4）。

\textbf{等价性类型}：精确等价（S-Procedure 对范数球上单个二次约束充要）。

\textbf{新变量}：$\mu_k \geq 0$（S-Procedure 松弛变量）。

\textbf{结果凸约束}：关于 $\mat{W}_k$ 和 $\mu_k$ 的线性矩阵不等式（LMI）。

\subsection{Step 4: SINR 分式线性化（消除 NC1）}

\textbf{变换}：交叉相乘消去分式。

原始分式形式（S-Procedure 后）：
\begin{equation}
\frac{\tr(\hat{\mat{H}}_k \mat{W}_k)}{\sum_{j\neq k} \tr(\hat{\mat{H}}_k \mat{W}_j) + \sigma_c^2} \geq \gamma_k
\end{equation}

其中 $\hat{\mat{H}}_k = \hat{\vect{h}}_k \hat{\vect{h}}_k^H$。

假设分母为正（因 $\sigma_c^2 > 0$ 恒满足）：
\begin{equation}
\tr(\hat{\mat{H}}_k \mat{W}_k) - \gamma_k \sum_{j\neq k} \tr(\hat{\mat{H}}_k \mat{W}_j) \geq \gamma_k \sigma_c^2 \tag{S4.1}
\end{equation}

\textbf{等价性类型}：严格等价（分母为正时）。

\textbf{结果凸约束}：关于 $\mat{W}_k$ 的线性不等式。

\subsection{Step 5a: PCRB 仿射展开（消除 NC6）}

\textbf{变换}：在单角度假设下线性化 FIM 迹。

在假设 1（$\nabla_{\boldsymbol{\theta}_p} \vect{g}_p$ 已知且恒定）下：
\begin{equation}
\mat{J}_p^{\text{data}} = \frac{2}{\sigma_s^2} \Real\Big\{ \nabla_{\boldsymbol{\theta}_p} \vect{g}_p^H \cdot \mat{R}_X \cdot \nabla_{\boldsymbol{\theta}_p}^H \vect{g}_p \Big\} \in \mathbb{C}^{D \times D}
\end{equation}
其中 $D$ 是目标状态维度（如 2D 平面 $D=2$）。迹循环 $\tr(\mat{A}\mat{B}\mat{C}) = \tr(\mat{C}\mat{A}\mat{B})$ 得：

\begin{align}
\tr(\mat{J}_p^{\text{data}}) &= \frac{2}{\sigma_s^2} \Real\Big\{ \tr\big(\nabla_{\boldsymbol{\theta}_p}^H \vect{g}_p \cdot \mat{R}_X \cdot \nabla_{\boldsymbol{\theta}_p} \vect{g}_p^H\big) \Big\} \notag \\
&= \frac{2}{\sigma_s^2} \Real\Big\{ \tr\big(\mat{R}_X \cdot \underbrace{\nabla_{\boldsymbol{\theta}_p} \vect{g}_p^H \cdot \nabla_{\boldsymbol{\theta}_p}^H \vect{g}_p}_{\in\, \mathbb{C}^{MN_t \times MN_t}}\big) \Big\}
\end{align}

定义 Hermitian PSD 常数矩阵
\begin{equation}
\mat{F}_p = \frac{2}{\sigma_s^2} \Real\Big\{ \nabla_{\boldsymbol{\theta}_p} \vect{g}_p^H \cdot \nabla_{\boldsymbol{\theta}_p}^H \vect{g}_p \Big\} \in \mathbb{H}^{MN_t}_+ \tag{S5.0}
\end{equation}
（注意：$\nabla_{\boldsymbol{\theta}_p} \vect{g}_p^H \in \mathbb{C}^{MN_t \times D}$、$\nabla_{\boldsymbol{\theta}_p}^H \vect{g}_p \in \mathbb{C}^{D \times MN_t}$，外积为 $\mathbb{C}^{MN_t \times MN_t}$；PSD 由 $\vect{x}^H (\mat{A}^H \mat{A}) \vect{x} = \|\mat{A}\vect{x}\|^2 \geq 0$ 保证）

则：
\begin{equation}
\tr(\mat{J}_p^{\text{data}}) = \tr(\mat{F}_p \mat{R}_X) \tag{S5.1}
\end{equation}

\textbf{等价性类型}：严格等价（在假设 1 下）。

\textbf{结果凸约束}：$\tr(\mat{F}_p \mat{R}_X) \geq \Gamma_{\text{Track},p}$ 对 $\mat{R}_X$ 线性。

\subsection{Step 5b: 感知 SINR 线性化}

\textbf{变换}：在提升域表达。

原始：$\text{SINR}_{S,p} = \frac{|\vect{g}_p^H \mat{Z} \vect{g}_p|}{\sigma_s^2} \geq \gamma_S^{\text{PoD}}$（一次方，参见 (P1-C2)）。

提升域（由 $\text{tr}(\vect{g}_p\vect{g}_p^H \mat{Z}) = \vect{g}_p^H \mat{Z} \vect{g}_p$）:
\begin{equation}
\tr(\vect{g}_p \vect{g}_p^H \mat{Z}) \geq \gamma_S^{\text{PoD}} \sigma_s^2 \tag{S5.2}
\end{equation}

\textbf{等价性类型}：严格等价（$\sigma_s^2 > 0$ 为常数，且 $\mat{Z} \succeq \mat{0} \Rightarrow \vect{g}_p^H \mat{Z} \vect{g}_p \geq 0$ 保证分子非负）。

\textbf{结果凸约束}：关于 $\mat{Z}$ 的线性不等式。

\subsection{Step 6: 功率约束（已凸）}

原始：$\sum_k \|\vect{w}_{m,k}\|^2 + \tr(\mat{Z}_m) \leq P_{\max}$。

提升域：
\begin{equation}
\tr(\mat{E}_m \mat{R}_X) \leq P_{\max} \tag{S6.1}
\end{equation}

其中 $\mat{E}_m$ 选择 AP $m$ 的天线。

\textbf{状态}：已凸（线性约束），无需变换。

\subsection{Step 7: AP 选择两步分解（消除 NC3）}

\textbf{变换}：分解为外层离散 + 内层连续。

原始：$\sum_m b_{mp} = N_{\text{req}}$，$b_{mp} \in \{0,1\}$。

\textbf{Step 7a（外层）}：基于大尺度衰落启发式选择 AP：
\begin{equation}
\mathcal{M}_p = \arg\max_{\mathcal{M} \subseteq \mathcal{M}, |\mathcal{M}| = N_{\text{req}}} \sum_{m \in \mathcal{M}} \text{PL}^{-1}(d_{m,p})
\end{equation}

\textbf{Step 7b（内层）}：固定 AP 集 $\mathcal{M}_p$ 求解凸 SDP。

\textbf{等价性类型}：工程启发式解（无理论最优性保证）。外层 AP 集合由大尺度衰落排序 top-$N_{\text{req}}$ 选取，**严格意义上既不保证下界也不保证上界**——选中的 AP 子集可能不是真正的最优组合（NP-hard），但内层 SDP 在固定 AP 集上仍为凸紧。

\textbf{结果}：多项式时间复杂度，次优但工程可接受。

\section{最终凸 SDP 问题}
%==============================================================================

综合所有步骤，原始非凸问题 (P1) 转化为如下标准凸 SDP：

\begin{align}
\min_{\{\mat{W}_k\}, \mat{Z}, \{\mu_k\}} \quad 
& \sum_{k=1}^K \tr(\mat{W}_k) + \tr(\mat{Z}) \tag{P3} \\
\text{s.t.} \quad 
& \begin{bmatrix} \mat{A}_k + \mu_k \mat{I} & \mat{A}_k \hat{\vect{h}}_k \\[1.5ex] \hat{\vect{h}}_k^H \mat{A}_k & \hat{\vect{h}}_k^H \mat{A}_k \hat{\vect{h}}_k - \sigma_c^2 - \mu_k \epsilon_h^2 \end{bmatrix} \succeq \mat{0}, \quad \forall k \tag{P3-C1} \\
& \tr(\vect{g}_p \vect{g}_p^H \mat{Z}) \geq \gamma_S^{\text{PoD}} \sigma_s^2, \quad \forall p \tag{P3-C2} \\
& \tr(\mat{F}_p \mat{R}_X) \geq \Gamma_{\text{Track},p}, \quad \forall p \tag{P3-C3} \\
& \tr(\mat{E}_m \mat{R}_X) \leq P_{\max}, \quad \forall m \tag{P3-C4} \\
& \mat{W}_k \succeq \mat{0}, \quad \forall k \tag{P3-C5} \\
& \mat{Z} \succeq \mat{0} \tag{P3-C6} \\
& \mu_k \geq 0, \quad \forall k \tag{P3-C7}
\end{align}

其中 $\mat{A}_k = \frac{1}{\gamma_k} \mat{W}_k - \sum_{j \neq k} \mat{W}_j$，$\mat{F}_p$ 在 Step 5a 中定义，$\mat{E}_m$ 为 AP 选择矩阵。

\begin{theorem}[问题凸性与等价性]
问题 (P3) 是标准凸 SDP。所有约束为 LMI 或线性不等式，目标线性。$K \leq 2$ 时 (P3) 与 (P1) 等价（SDR 紧致）。$K > 2$ 时 (P3) 提供 (P1) 下界。(P3) 可通过内点法在多项式时间求解，复杂度上界为 $O\big((K+1)^3 (MN_t)^6 \cdot (K+P+M)\big)$——精确源于 (P3) 决策变量维数 $n = O(K (MN_t)^2)$，LMI 约束最大尺寸 $(MN_t+1)$；简化量级 $O((K M N_t^2)^{3.5})$。
\end{theorem}

\begin{proof}
(P3) 中各约束为 LMI (P3-C1)、线性不等式 (P3-C2, P3-C3, P3-C4) 或 PSD 锥约束 (P3-C5, P3-C6)。目标线性。故 (P3) 为凸 SDP。Steps 1, 3, 4, 5a, 5b, 6 为严格等价。Step 2 在 $K \leq 2$ 时为紧致松弛。Step 7 为启发式近似（外层 AP 集合由大尺度衰落排序选取，\textbf{无理论最优性保证}——既可能给出原问题下界，也可能因启发式选错 AP 而给出不可比较的目标值）。故 $K \leq 2$ 时 (P3) $\equiv$ (P1)；$K > 2$ 时 $P_{\text{P3}}^* \leq P_{\text{P1}}^*$。复杂度上界由内点法对 SDP 的标准结论给出（Nesterov-Todd 路径跟踪，最坏情况 $O(n^3 L)$，其中 $n$ 为决策变量维数，$L$ 为输入数据位长）。
\end{proof}
